 "bytes"; "context"; "fmt"; "io"; "mime/multipart"; "net/http"; "net/http/httptest"; "os"; "path/filepath"; "strings"

func must(e error) { if e != nil { panic(e) } }
func list(p string) string { x,e:=os.ReadDir(p); must(e); a:=[]string{}; for _,v:=range x { if strings.HasPrefix(v.Name(),"body") { a=append(a,"body") } else if strings.HasPrefix(v.Name(),"crzmp") { a=append(a,"crzmp") } }; return fmt.Sprint(a) }
func body(n int) ([]byte,string) { var b bytes.Buffer; w:=multipart.NewWriter(&b); p,e:=w.CreateFormFile("f","a.txt"); must(e); _,e=p.Write(bytes.Repeat([]byte("x"),n)); must(e); must(w.Close()); return b.Bytes(),w.FormDataContentType() }
func direct(name, dir, alt string, n int, fail bool) { d:="SecRuleEngine On\nSecRequestBodyAccess On\nSecRequestBodyLimit 1048576\nSecRequestBodyInMemoryLimit 1024\n"; if alt!="" {d+="SecUploadDir "+alt+"\n"}; engine,e:=coraza.NewWAF(coraza.NewWAFConfig().WithDirectives(d)); must(e); tx:=engine.NewTransaction(); tx.ProcessConnection("127.0.0.1",1,"127.0.0.1",80); tx.ProcessURI("/upload","POST","HTTP/1.1"); b,ct:=body(n); tx.AddRequestHeader("Content-Type",ct); if it:=tx.ProcessRequestHeaders(); it!=nil {panic(it)}; if name=="spill_create_fault" {must(os.Chmod(dir,0500))}; it,wn,e:=tx.WriteRequestBody(b); if name=="spill_create_fault" {must(os.Chmod(dir,0700))}; fmt.Printf("%s write=%d/%d interruption=%v error=%v before=%s\n",name,wn,len(b),it,e,list(dir)); if e==nil { if fail {must(os.Chmod(dir,0500))}; it,e=tx.ProcessRequestBody(); v:=tx.(plugintypes.TransactionState).Variables(); fmt.Printf("%s process interruption=%v error=%v reqbody_error=%s engine=%s alternate=%s\n",name,it,e,v.RequestBodyError().Get(),list(dir),list(filepath.Join(filepath.Dir(dir),"alternate"))); if fail {must(os.Chmod(dir,0700))} }; fmt.Printf("%s close=%v after=%s\n",name,tx.Close(),list(dir)) }
type flip struct { r *bytes.Reader; dir string; done bool }
func (f *flip) Read(b []byte)(int,error) { n,e:=f.r.Read(b); if e==io.EOF&&!f.done { must(os.Chmod(f.dir,0500)); f.done=true }; return n,e }
func (f *flip) Close() error {return nil}
func jul(name,dir,mode string,crs bool) { cfg:=config.WAFConfig{Enabled:true,Mode:mode,CRSEnabled:crs,RequestBodyLimit:config.Size(1<<20), InlineRules:"SecRequestBodyAccess On"}; fw,e:=waf.New(context.Background(),cfg,waf.Options{}); must(e); defer fw.Close(); b,ct:=body(128); r:=httptest.NewRequest("POST","http://example.test/upload",nil); r.Body=&flip{r:bytes.NewReader(b),dir:dir}; r.ContentLength=int64(len(b)); r.Header.Set("Content-Type",ct); reached:=false; count:=int64(0); h:=fw.Middleware()(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){reached=true; count,_=io.Copy(io.Discard,r.Body); w.WriteHeader(204)})); rr:=httptest.NewRecorder(); h.ServeHTTP(rr,r); must(os.Chmod(dir,0700)); fmt.Printf("%s mode=%s crs=%t status=%d origin=%t bytes=%d\n",name,mode,crs,rr.Code,reached,count) }
func lateCleanup(dir string) { d:="SecRuleEngine On\nSecRequestBodyAccess On\nSecRequestBodyLimit 1048576\nSecRequestBodyInMemoryLimit 1024\n"; engine,e:=coraza.NewWAF(coraza.NewWAFConfig().WithDirectives(d)); must(e); tx:=engine.NewTransaction(); tx.ProcessConnection("127.0.0.1",1,"127.0.0.1",80); tx.ProcessURI("/upload","POST","HTTP/1.1"); tx.AddRequestHeader("Content-Type","text/plain"); if it:=tx.ProcessRequestHeaders(); it!=nil {panic(it)}; _,_,e=tx.WriteRequestBody(bytes.Repeat([]byte("x"),4096)); must(e); _,e=tx.ProcessRequestBody(); must(e); files,e:=filepath.Glob(filepath.Join(dir,"body*")); must(e); if len(files)!=1 {panic("expected one owned body file")}; must(os.Remove(files[0])); e=tx.Close(); fmt.Printf("late_cleanup_fault close_error=%t error_is_not_exist=%t\n",e!=nil,os.IsNotExist(e)) }
func main(){ dir:=os.Getenv("TMPDIR"); alt:=filepath.Join(filepath.Dir(dir),"alternate"); if os.Getenv("PROBE_NOFS") == "1" { direct("nofs_small_multipart",dir,"",128,false); direct("nofs_spill",dir,"",4096,false); return }; direct("normal",dir,"",4096,false); direct("explicit_upload_dir",dir,alt,4096,false); direct("multipart_create_fault",dir,"",128,true); direct("spill",dir,"",4096,false); direct("spill_create_fault",dir,"",4096,false); lateCleanup(dir); jul("jul_create_fault",dir,"block",false); jul("jul_create_fault",dir,"block",true); jul("jul_create_fault",dir,"detect",true) }
